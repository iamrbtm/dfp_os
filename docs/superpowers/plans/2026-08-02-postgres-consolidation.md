# Postgres Consolidation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Move DFPos from MariaDB plus two Postgres services to one Postgres relational database engine for the Flask app, audit-log service, and intelligence service, while keeping Redis for Celery and service coordination.

**Architecture:** The app will use Postgres as the primary SQLAlchemy database through `psycopg`, with the audit-log and intelligence services kept on the same Postgres engine but isolated by separate databases. Docker Compose will expose one Postgres service and create three databases/users through deterministic init SQL. Data migration will be performed through a scripted export/import path that can be rehearsed against a copy before replacing the live MariaDB-backed app.

**Tech Stack:** Python 3.14, Flask, SQLAlchemy ORM, Flask-Migrate/Alembic, Postgres 17, psycopg 3, asyncpg for async services, Docker Compose, Redis, Pytest, Ruff.

## Global Constraints

- Python runtime remains `3.14`.
- Use `uv` for dependency management and command execution.
- MariaDB is replaced only after Postgres app tests, migration smoke tests, and Docker health checks pass.
- Keep Redis for Celery broker/result backend unless a separate queue replacement plan is approved.
- Do not remove audit-log service isolation; consolidate its database engine, not its service boundary.
- Do not store secrets in committed files; `.env.example` may contain placeholders only.
- Use `Numeric`/`Decimal`/integer cents for money; do not introduce floats for money.
- Preserve audit event source-of-truth semantics in the audit-log service.
- Commit after every task that reaches a green targeted verification gate.

```python
    admin_url = admin_database_url()
    database_name = _database_name(database_url)
    username = _database_user(database_url)
    password = (_database_password(database_url) or "")
    engine = create_engine(admin_url, isolation_level="AUTOCOMMIT")

            exists = connection.execute(
                text("SELECT 1 FROM pg_database WHERE datname = :name"),
                {"name": database_name},

            if not username:

            role_exists = connection.execute(

                {"name": username},
            ).scalar()
            if not role_exists:

                    {"password": password},
                )
            if not exists:
```

```python
from __future__ import annotations

import os
from collections.abc import Iterable

from sqlalchemy import MetaData, Table, create_engine, select


KNOWN_TABLE_ORDER = [
    "businesses",
    "users",
    "categories",
    "collections",
    "products",
    "product_collection_memberships",
    "product_images",
    "product_launch_checklist_items",
    "product_photo_shots",
    "inventory_locations",
    "inventory_records",
    "orders",
    "order_items",
    "payments",
]


def ordered_tables(table_names: Iterable[str]) -> list[str]:
    names = set(table_names)
    known = [name for name in KNOWN_TABLE_ORDER if name in names]
    unknown = sorted(names.difference(known))
    return known + unknown


def copy_table(source_engine, target_engine, table_name: str) -> int:
    source_meta = MetaData()
    target_meta = MetaData()
    source_table = Table(table_name, source_meta, autoload_with=source_engine)
    target_table = Table(table_name, target_meta, autoload_with=target_engine)
    rows = []
    with source_engine.connect() as source_conn:
        for row in source_conn.execute(select(source_table)):
            rows.append(dict(row._mapping))
    if not rows:
        return 0
    with target_engine.begin() as target_conn:
        target_conn.execute(target_table.insert(), rows)
    return len(rows)


def main() -> int:
    source_url = os.environ["SOURCE_DATABASE_URL"]
    target_url = os.environ["TARGET_DATABASE_URL"]
    source_engine = create_engine(source_url)
    target_engine = create_engine(target_url)
    source_meta = MetaData()
    source_meta.reflect(bind=source_engine)
    for table_name in ordered_tables(source_meta.tables.keys()):
        count = copy_table(source_engine, target_engine, table_name)
        print(f"{table_name}: {count}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
def reset_postgres_sequences(target_engine) -> None:
    sql = """
    SELECT
      quote_ident(sequence_schema) || '.' || quote_ident(sequence_name) AS sequence_name,
      quote_ident(table_schema) || '.' || quote_ident(table_name) AS table_name,
      quote_ident(column_name) AS column_name

    JOIN information_schema.columns
      ON column_default = 'nextval(''' || sequence_schema || '.' || sequence_name || '''::regclass)'

    """
    with target_engine.begin() as conn:
        for row in conn.execute(text(sql)):
            conn.execute(
                text(
                    f"SELECT setval('{row.sequence_name}', "
                    f"COALESCE((SELECT MAX({row.column_name}) FROM {row.table_name}), 1), true)"
                )
            )
```

Call `reset_postgres_sequences(target_engine)` at the end of `main()`.

- If writes occurred after cutover, stop the app, export changed Postgres rows, and reconcile manually before reverting. Do not discard either database volume.
